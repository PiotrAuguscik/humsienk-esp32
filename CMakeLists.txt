cmake_minimum_required(VERSION 3.16)
project(humsienk_esp32 LANGUAGES CXX)
option(HUMSIENK_SANITIZE "Enable undefined-behavior sanitizer for host tests" OFF)
add_library(humsienk src/humsienk.cpp)
target_include_directories(humsienk PUBLIC src)
target_compile_features(humsienk PUBLIC cxx_std_11)
target_compile_options(humsienk PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(HUMSIENK_SANITIZE)
  target_compile_options(humsienk PUBLIC -fsanitize=undefined -fno-omit-frame-pointer)
  target_link_options(humsienk PUBLIC -fsanitize=undefined)
endif()
enable_testing()
add_executable(humsienk_tests tests/test_main.cpp)
target_link_libraries(humsienk_tests PRIVATE humsienk)
target_compile_options(humsienk_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
add_test(NAME protocol_and_session COMMAND humsienk_tests)
